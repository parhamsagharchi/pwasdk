// Listen to device motion and spatial orientation

  const { alpha, beta, gamma } = e;
  card.style.transform = `rotateX(${beta}deg) rotateY(${gamma}deg)`;
});

                  
import { useEffect, useState } from 'react';

export function TiltCard() {
  const [rot, setRot] = useState({ x: 0, y: 0 });

  useEffect(() => {
    const handleMove = (e) => {
      setRot({ x: e.beta || 0, y: e.gamma || 0 });
    };
    window.addEventListener('deviceorientation', handleMove);
    return () => window.removeEventListener('deviceorientation', handleMove);
  }, []);

  return <div style={{ transform: `rotateX(${rot.x}deg) rotateY(${rot.y}deg)` }} />;
}

                  
window.addEventListener('deviceorientation', (e) => {
  console.log('Angles:', e.alpha, e.beta, e.gamma);
});
